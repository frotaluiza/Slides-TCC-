\begin{slide}{Abordagens de Modelagem em MD}
  \small
  \centering
  \resizebox{\textwidth}{!}{
    \begin{tabular}{p{0.18\linewidth} p{0.35\linewidth} p{0.18\linewidth} p{0.22\linewidth}}
      \toprule
      \textbf{Abordagem} & \textbf{Descrição} & \textbf{Vantagens} & \textbf{Limitações} \\
      \midrule
      \textbf{Física}
      & Modelos 0D, 1D, 2D
      & Interpretabilidade física
      & Custo computacional cresce com a complexidade (0D $\rightarrow$ 1D $\rightarrow$ 2D) \\
      \midrule
      \textbf{Dados}
      & Técnicas de regressão (Ex: Redes Neurais) para predição de fluxo e temperaturas
      & Alta flexibilidade; aprox. não linear
      & Pouca interpretabilidade; generalizam bem com muitos dados e mal quando há escassez de dados fora da região de treinamento \\
      \midrule
      \textbf{Híbridos (físico + dados)}
      & Combinam modelos físicos reduzidos com técnicas de regressão
      & Combina física + dados; regulariza o espaço de busca
      & Complexidade de implementação \\
      \bottomrule
    \end{tabular}
  }
  \note{A revisão mostra três grandes abordagens para modelagem em MD. Física: interpretável mas cara — custo computacional cresce com a complexidade. Data-driven: flexível, mas com pouca interpretabilidade e generalização dependente do volume de dados — mal quando há escassez de dados fora da região de treinamento. Híbridos: combinam o melhor dos dois mundos com maior complexidade de implementação. [~1,5 min]}
\end{slide}

\begin{slide}{Lacunas e Contribuições}
  \scriptsize
  \begin{columns}
    \begin{column}{.5\linewidth}
      \textbf{Lacunas na literatura:}
      \begin{Item}
        \item[\textbf{1.}] \textbf{Ausência de validação cruzada por grupos} --- partição única ignora grupos experimentais, mascara sobreajuste e produz estimativas otimistas. Risco agravado pelo volume limitado de dados em escala piloto \cite{Vabalas2019}
        \item[\textbf{2.}] \textbf{Seleção por RMSE puro favorece superparametrização} --- ignora a variabilidade estatística entre partições. A estratégia do \textit{one-standard-error} (1-SE) \cite{Hastie2009, Yates2022} \textit{(navalha de Occam)} é um critério estabelecido de escolha de modelos: dentro do conjunto de modelos estatisticamente equivalentes ao melhor (erro dentro de $\pm 1$ erro padrão da validação cruzada), seleciona-se o de menor complexidade
        \item[\textbf{3.}] \textbf{Modelos híbridos (físico+ML) em MD são praticamente inexistentes} --- apenas 1 trabalho aplica PINN em DCMD; nenhum trabalho em V-AGMD
      \end{Item}
    \end{column}
    \begin{column}{.5\linewidth}
      \textbf{Contribuições do trabalho:}
      \begin{Item}
        \item Validação cruzada (regimes de salinidade)
        \item Critério 1-SE: estratégia de escolha de modelos menos complexos dentro de um desvio padrão do erro da validação cruzada \cite{Hastie2009ESL}
        \item Implementação de arquiteturas híbridas pré-existentes \cite{Rai2020} (4 variantes) aplicadas ao V-AGMD
      \end{Item}
    \end{column}
  \end{columns}
  \vfill
  {\scriptsize Nota: o critério 1-SE, a validação cruzada por grupos e a hibridização são estratégias pré-existentes \cite{Hastie2009, Yates2022, Rai2020}; a contribuição do trabalho é a aplicação sistemática ao V-AGMD.}
  \note{Lacunas: 1) validação por partição única, sem CV por grupos — métricas otimistas para extrapolação, risco de sobreajuste. 2) seleção por RMSE puro ignora variabilidade — 1-SE (navalha de Occam, \cite{Hastie2009, Yates2022}) é estratégia estabelecida que escolhe o modelo mais simples dentro de 1 erro padrão do melhor. 3) híbridos em MD praticamente inexistentes. Contribuições: aplicação sistemática de estratégias pré-existentes (validação cruzada, 1-SE, arquiteturas híbridas da literatura) ao V-AGMD. [~2 min]}
\end{slide}
